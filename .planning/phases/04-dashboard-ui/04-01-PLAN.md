    Create `cli/src/ui/analytics/hooks/useThrottle.ts` with a custom throttle hook using lodash.debounce:

    ```typescript
    import { useEffect, useMemo, useState } from "react"
    import { debounce } from "lodash.debounce"

    export function useThrottle<T>(value: T, delay: number): T {
      const [throttledValue, setThrottledValue] = useState<T>(value)

      const debouncedSetValue = useMemo(
        () => debounce((newValue: T) => setThrottledValue(newValue), delay),
        [delay]
      )

      useEffect(() => {
        debouncedSetValue(value)
        return () => debouncedSetValue.cancel()
      }, [value, debouncedSetValue])

      return throttledValue
    }
    ```

    Key requirements:
    - Generic type parameter `<T>` to work with any data type
    - Uses useMemo to stabilize debounce function across re-renders
    - Cleanup function cancels pending debounced calls on unmount
    - Default delay of 100-150ms for dashboard use cases

    Reference: Research document section "Pattern 3: Custom useThrottle Hook"

  </action>
