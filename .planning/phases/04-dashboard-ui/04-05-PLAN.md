
    **2. Add keybinding in keyboard handler:**
    Find the main keyboard input handler (useEffect with input.on("keypress")) and add these bindings in the keypress switch statement:

    ```typescript
    // Ctrl+S: Toggle analytics dashboard
    if (key.ctrl && key.name === "s") {
        const store = getDefaultStore()
        const currentShowDashboard = store.get(showDashboardAtom)
        store.set(showDashboardAtom, !currentShowDashboard)
        return
    }

    // Escape: Exit dashboard view (if active)
    if (key.name === "escape") {
        const store = getDefaultStore()
        const currentShowDashboard = store.get(showDashboardAtom)
        if (currentShowDashboard) {
            store.set(showDashboardAtom, false)
            return
        }
    }
    ```

    Place these bindings early in the switch statement, after existing Ctrl+C but before other key combinations.
